# DÆMON-KALSHI runtime dependencies ONLY.
#
# Separate from requirements.txt because that file serves both bots in this
# repo, and installing the Polymarket stack (py-clob-client, polymarket-us,
# web3 and their transitive chain) on the Kalshi service is a slower build
# and a larger surface for a dependency conflict to break a deploy — for
# packages the Kalshi daemon never imports.
#
# Every entry below is imported by core/, workers/, memory/ or main.py.
# Verified with:
#   grep -rhoE '^(import|from) [a-z_]+' core/ workers/ memory/ main.py config.py
#
# Versions are pinned to a tested minor range rather than left open. An
# unpinned floor means a redeploy months from now silently installs a
# different library than the one this was tested against — on a bot that
# moves real money, "it worked last time" is not a build strategy.

# Kalshi REST/WS client, all HTTP in the repo (ESPN, NOAA, FRED, CoinGecko,
# Yahoo, Telegram, Moonshot).
httpx>=0.27,<0.29

# RSA-PSS request signing for Kalshi auth.
cryptography>=42.0,<46

# Checker (Claude).
anthropic>=0.40,<1.0

# Kalshi market-data WebSocket (core/kalshi_ws.py).
websockets>=12.0,<16

# Loads .env in local development. On Railway real env vars are already
# injected and config.py degrades gracefully if this is missing.
python-dotenv>=1.0,<2.0
